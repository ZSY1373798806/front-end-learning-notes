### 统计源码里面的 console.log 调用数量与调用路径

- 定义插件类

  定义一个JavaScript类；在类的apply方法中，监听webpack的compilation钩子来访问处理模块的源代码

- 监听适当的webpack钩子

  针对源代码的处理，选择监听compilation阶段的optimizeModule钩子；在这个阶段，模块的源代码可以被访问和修改

- 处理源代码

  处理每个模块的源代码，可以使用简单的正则表达式来识别console.log的调用

```js
class ConsoleLogStatsPlugin {
  constructor(options = {}) {
    this.options = {
      showAll: false, // 是否显示所有模块（包括没有 console.log 的）
      threshold: 0,  // 只显示超过此数量的模块
      ...options
    };
    this.stats = [];
  }
  apply(compiler) {
    compiler.hooks.compilation.tap("ConsoleLogStatsPlugin", (compilation) => {
      compilation.moduleTemplates.javascript.hooks.render.tap("ConsoleLogStatsPlugin", (moduleSource, module) => {
        if (!module.resource) return moduleSource;
        const source = moduleSource.source();
        const consoleLogMatches = source.match(/console\.(log|info|warn|error)\(/g) || [];
        const count = consoleLogMatches.length;
        if (count > this.options.threshold || this.options.showAll) {
          this.stats.push({
            path: module.resource,
            count,
            warnings: consoleLogMatches.filter(c => c.includes('.warn')).length,
            errors: consoleLogMatches.filter(c => c.includes('.error')).length,
            logs: consoleLogMatches.filter(c => c.includes('.log')).length,
            infos: consoleLogMatches.filter(c => c.includes('.info')).length
          });
        }
        return moduleSource;
      });
      // 在编译完成后输出统计结果
      compilation.hooks.afterProcessAssets.tap("ConsoleLogStatsPlugin", () => {
        if (this.stats.length > 0) {
          const total = this.stats.reduce((sum, item) => sum + item.count, 0);
          console.log(`\n📊 Console Log Statistics (${total} calls in ${this.stats.length} modules):`);
          this.stats
            .sort((a, b) => b.count - a.count)
            .forEach(item => {
              const relativePath = item.path.replace(process.cwd(), '');
              const typeStats = [
                item.logs > 0 ? `log: ${item.logs}` : '',
                item.infos > 0 ? `info: ${item.infos}` : '',
                item.warnings > 0 ? `warn: ${item.warnings}` : '',
                item.errors > 0 ? `error: ${item.errors}` : ''
              ].filter(Boolean).join(', ');
              console.log(`  ${relativePath}: ${item.count} calls (${typeStats})`);
            });
        } else {
          console.log('\n✅ No console.log calls found in any modules.');
        }
      });
    });
  }
}
module.exports = ConsoleLogStatsPlugin;
```



### 获取登录token的webpack插件

```js
import type { Compiler, Compilation, WebpackError, Stats } from 'webpack';
import type { InnerCallback } from 'tapable';

/**
 * 登录接口响应数据结构
 */
interface LoginResponse {
  code: number;          // 响应状态码，0表示成功
  data: any;            // 响应数据，包含token信息
  msg?: string;         // 错误消息
  message?: string;     // 备用错误消息字段
}

/**
 * 插件配置选项
 */
interface PluginOptions {
  url: string;                              // 登录接口URL
  params: Record<string, any>;              // 登录请求参数（用户名、密码等）
  tokenKeyName?: string;                    // token在响应数据中的字段名，默认'token'
  localStorageKey?: string;                 // 存储到localStorage的key名，默认'token'
  extraObj?: Record<string, any>;           // 需要额外合并到token数据的对象
  timeout?: number;                         // 请求超时时间（毫秒），默认10秒
  retryCount?: number;                      // 失败重试次数，默认3次
  enableInProduction?: boolean;             // 是否在生产环境启用，默认false
}

class LoginWebpackPlugin {
  private readonly options: Required<PluginOptions>;
  private tokenData: any = null;
  private isLoginCalled = false;

  constructor(options: PluginOptions) {
    // 参数验证
    this.validateOptions(options);
    
    // 设置默认值
    this.options = {
      tokenKeyName: 'token',
      localStorageKey: 'token',
      timeout: 10000,
      retryCount: 3,
      enableInProduction: false,
      extraObj: {},
      ...options
    };
  }

  /**
   * 验证插件配置参数
   * @param options 用户传入的配置选项
   * @throws {Error} 参数不合法时抛出错误
   */
  private validateOptions(options: PluginOptions): void {
    // 检查必需的URL参数
    if (!options.url) {
      throw new Error('LoginWebpackPlugin: url参数是必需的');
    }
    
    // 检查必需的请求参数
    if (!options.params || typeof options.params !== 'object') {
      throw new Error('LoginWebpackPlugin: params必须是一个有效的对象');
    }
    
    // 验证URL格式是否正确
    try {
      new URL(options.url);
    } catch {
      throw new Error('LoginWebpackPlugin: url必须是一个有效的URL地址');
    }
  }

  /**
   * 调用登录接口获取token
   * 包含重试机制和错误处理
   * @throws {Error} 登录失败时抛出错误
   */
  private async callLoginApi(): Promise<void> {
    const { url, params, timeout, retryCount } = this.options;
    
    let lastError: Error | null = null;
    
    // 实现重试机制：最多重试指定次数
    for (let attempt = 1; attempt <= retryCount; attempt++) {
      try {
        console.log(`🔄 第${attempt}次尝试登录...`);
        
        // 发起HTTP请求
        const response = await this.makeRequest(url, params, timeout);
        
        // 处理登录响应
        this.handleLoginResponse(response);
        
        console.log('✅ 登录成功！');
        return; // 成功则直接返回
      } catch (error) {
        lastError = error as Error;
        console.warn(`⚠️  第${attempt}/${retryCount}次登录尝试失败:`, error.message);
        
        // 如果不是最后一次尝试，等待后重试
        if (attempt < retryCount) {
          // 指数退避策略：等待时间随尝试次数增加
          const delayMs = Math.pow(2, attempt - 1) * 1000;
          console.log(`⏳ ${delayMs}ms后进行重试...`);
          await this.delay(delayMs);
        }
      }
    }
    
    // 所有重试都失败，抛出最后的错误
    throw new Error(`❌ 登录失败，已重试${retryCount}次: ${lastError?.message}`);
  }

  /**
   * 发起HTTP登录请求
   * @param url 登录接口地址
   * @param params 请求参数
   * @param timeout 超时时间（毫秒）
   * @returns Promise<LoginResponse> 登录响应数据
   * @throws {Error} 网络错误或超时时抛出错误
   */
  private async makeRequest(url: string, params: any, timeout: number): Promise<LoginResponse> {
    // 创建AbortController用于超时控制
    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), timeout);

    try {
      // 发起fetch请求
      const response = await fetch(url, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'User-Agent': 'LoginWebpackPlugin/1.0'  // 标识请求来源
        },
        body: JSON.stringify(params),              // 序列化请求体
        signal: controller.signal                  // 绑定取消信号
      });

      // 清除超时定时器
      clearTimeout(timeoutId);

      // 检查HTTP响应状态
      if (!response.ok) {
        throw new Error(`HTTP请求失败 ${response.status}: ${response.statusText}`);
      }

      // 解析JSON响应
      const data = await response.json();
      return data;
      
    } catch (error) {
      clearTimeout(timeoutId);
      
      // 处理特定的超时错误
      if (error.name === 'AbortError') {
        throw new Error(`请求超时，超过${timeout}ms未响应`);
      }
      
      // 重新抛出其他错误
      throw error;
    }
  }

  /**
   * 处理登录接口响应数据
   * @param response 登录接口返回的响应数据
   * @throws {Error} 登录失败或数据格式错误时抛出错误
   */
  private handleLoginResponse(response: LoginResponse): void {
    const { code, data, msg, message } = response;
    
    // 检查业务状态码
    if (code !== 0) {
      const errorMsg = msg || message || '登录接口返回错误状态';
      throw new Error(`登录失败: ${errorMsg}`);
    }

    const { tokenKeyName } = this.options;
    const tokenValue = data?.[tokenKeyName];
    
    // 验证token数据是否存在
    if (!tokenValue) {
      throw new Error(`响应数据中未找到token字段: data.${tokenKeyName}`);
    }

    // 安全地合并token数据和额外数据
    this.tokenData = this.mergeTokenData(tokenValue, this.options.extraObj);
    
    console.log('✅ 登录接口调用成功，token数据已获取');
  }

  /**
   * 合并token数据和额外对象
   * @param tokenValue 从登录接口获取的原始token数据
   * @param extraObj 需要额外合并的对象数据
   * @returns 合并后的最终数据
   */
  private mergeTokenData(tokenValue: any, extraObj: Record<string, any>): any {
    // 如果没有额外数据需要合并，直接返回token值
    if (Object.keys(extraObj).length === 0) {
      return tokenValue;
    }

    // 如果token是字符串，需要包装成对象才能合并
    const baseTokenObj = typeof tokenValue === 'string' 
      ? { [this.options.tokenKeyName]: tokenValue }  // {token: "abc123"}
      : tokenValue;                                   // 已经是对象，直接使用

    // 使用展开运算符合并对象，extraObj会覆盖同名属性
    return { ...baseTokenObj, ...extraObj };
  }

  /**
   * 生成注入到HTML中的JavaScript脚本
   * 该脚本会在浏览器中执行，将token数据存储到localStorage
   * @returns 包含完整script标签的HTML字符串
   */
  private generateInjectionScript(): string {
    const { localStorageKey } = this.options;
    const serializedData = JSON.stringify(this.tokenData);
    
    // 生成安全的JavaScript代码，防止XSS攻击
    return `
      <script>
        // 使用立即执行函数包装，避免污染全局作用域
        (function() {
          try {
            // 获取token数据（已经过JSON序列化处理）
            const tokenData = ${serializedData};
            
            // 根据数据类型决定存储格式
            const storageValue = typeof tokenData === 'string' 
              ? tokenData                    // 字符串直接存储
              : JSON.stringify(tokenData);   // 对象需要序列化
            
            // 存储到localStorage
            window.localStorage.setItem(${JSON.stringify(localStorageKey)}, storageValue);
            
            // 触发自定义事件，通知应用token已准备就绪
            // 应用可以监听此事件来初始化需要token的功能
            window.dispatchEvent(new CustomEvent('tokenReady', { 
              detail: { tokenData } 
            }));
            
            console.log('🔐 Token数据注入成功');
          } catch (error) {
            console.error('❌ Token注入失败:', error);
          }
        })();
      </script>`;
  }

  /**
   * 延迟执行工具函数
   * @param ms 延迟时间（毫秒）
   * @returns Promise 延迟完成后resolve
   */
  private delay(ms: number): Promise<void> {
    return new Promise(resolve => setTimeout(resolve, ms));
  }

  /**
   * 判断是否应该注入token
   * 根据环境和配置决定是否执行token注入
   * @returns boolean 是否应该注入
   */
  private shouldInjectToken(): boolean {
    const isDevelopment = process.env.NODE_ENV === 'development';
    
    // 开发环境默认启用，生产环境需要明确配置
    return isDevelopment || this.options.enableInProduction;
  }

  /**
   * Webpack插件的入口方法
   * 注册到Webpack的各个生命周期钩子
   * @param compiler Webpack编译器实例
   */
  apply(compiler: Compiler): void {
    const pluginName = 'LoginWebpackPlugin';

    // 钩子1: 在编译开始前获取token
    // 这确保了在构建过程开始前就完成登录流程
    compiler.hooks.beforeCompile.tapAsync(
      pluginName,
      async (params, callback) => {
        // 防止重复调用登录接口
        if (this.isLoginCalled) {
          return callback();
        }

        try {
          console.log('🚀 开始执行登录流程...');
          
          // 异步调用登录接口
          await this.callLoginApi();
          
          // 标记已调用，防止重复执行
          this.isLoginCalled = true;
          
          // 通知Webpack继续编译流程
          callback();
        } catch (error) {
          console.error('❌ 登录流程失败:', error.message);
          
          // 将错误传递给Webpack，中断构建过程
          callback(error as WebpackError);
        }
      }
    );

    // 钩子2: 在compilation阶段注入token到HTML
    // 这个阶段可以修改HTML内容
    compiler.hooks.compilation.tap(pluginName, (compilation: Compilation) => {
      // 检查并获取HtmlWebpackPlugin
      let HtmlWebpackPlugin: any;
      try {
        HtmlWebpackPlugin = require('html-webpack-plugin');
      } catch {
        console.warn('⚠️  未找到HtmlWebpackPlugin，跳过token注入');
        return;
      }

      // 获取HtmlWebpackPlugin的钩子
      const hooks = HtmlWebpackPlugin.getHooks(compilation);
      
      // 在HTML模板执行完成后注入token脚本
      hooks.afterTemplateExecution.tapAsync(
        pluginName,
        (htmlPluginData: { html: string }, callback: Callback<{ html: string }>) => {
          try {
            // 检查是否需要注入token
            if (this.shouldInjectToken() && this.tokenData) {
              console.log('💉 正在注入token数据到HTML...');
              
              // 在HTML末尾添加token注入脚本
              htmlPluginData.html += this.generateInjectionScript();
              
              console.log('✅ Token脚本注入完成');
            } else {
              console.log('ℹ️  跳过token注入（环境限制或无token数据）');
            }
            
            // 通知HtmlWebpackPlugin继续处理
            callback(null, htmlPluginData);
          } catch (error) {
            console.error('❌ Token注入过程发生错误:', error);
            
            // 即使注入失败，也要继续构建流程，但传递错误信息
            callback(error as WebpackError, htmlPluginData);
          }
        }
      );
    });

    // 钩子3: 构建完成后的总结日志
    // 显示插件执行结果和token信息摘要
    compiler.hooks.done.tapAsync(
      pluginName,
      (stats: Stats, callback: InnerCallback<Error, void>) => {
        // 检查构建是否有错误
        const hasErrors = stats.compilation.errors.length > 0;
        
        if (!hasErrors && this.tokenData) {
          // 生成token数据摘要（避免敏感信息泄露）
          const tokenSummary = typeof this.tokenData === 'string' 
            ? `"${this.tokenData.substring(0, 20)}..."${this.tokenData.length > 20 ? ` (长度:${this.tokenData.length})` : ''}`
            : `{${Object.keys(this.tokenData).join(', ')}}`;
            
          console.log('✅ 登录插件执行完成');
          console.log(`🔐 Token数据摘要: ${tokenSummary}`);
          console.log(`📦 数据已存储到localStorage.${this.options.localStorageKey}`);
        } else if (hasErrors) {
          console.log('⚠️  构建过程中发现错误，请检查日志');
        } else {
          console.log('ℹ️  插件执行完成，但未获取到token数据');
        }
        
        // 通知Webpack插件执行完成
        callback();
      }
    );
  }
}

// 导出插件类供外部使用
export = LoginWebpackPlugin;
```
